# --- Apache TVM (not pip-resolvable at main; built from source) -------------
# The reference environment uses apache/tvm main @ 7e06fc6c1420d0188eb9d889bd74e2e1fb76e448
# Keep its vendored FFI headers: newer headers rename constants used by this TVM revision.
#   git clone --recursive https://github.com/apache/tvm.git <tvm>
#   git -C <tvm> checkout 7e06fc6c1420d0188eb9d889bd74e2e1fb76e448
#   git -C <tvm> submodule update --init --recursive
#   mkdir -p <tvm>/build
#   cp <tvm>/cmake/config.cmake <tvm>/build/   # then edit USE_LLVM / USE_CUDA etc.
#   cmake -G Ninja -S <tvm> -B <tvm>/build && cmake --build <tvm>/build
#   pip install --no-build-isolation <tvm> \
#     -Ccmake.define.TVM_PACKAGE_EXTRA_LIBS="<tvm>/build/lib/libtvm_runtime_cuda.so;<tvm>/build/lib/libtvm_runtime_extra.so"
# (drop TVM_PACKAGE_EXTRA_LIBS for a CPU-only build).  Installing as a regular
# wheel puts the shared libraries in site-packages/tvm/lib, where build.rs finds
# them; an editable install (build/lib) is found too.
